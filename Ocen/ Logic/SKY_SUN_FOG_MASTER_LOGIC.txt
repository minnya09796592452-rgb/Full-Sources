	private bool CheckUpdate()
	{
		if (Options.GraphicsSettings.ShadowQuality > Quality_Shadows.MEDIUM)
		{
			return true;
		}
		float militaryTime = Singleton<GameTime>.Instance.MilitaryTime;
		if ((double)_lastUpdatedTime < 0.0)
		{
			_lastUpdatedTime = militaryTime;
			return true;
		}
		if (Mathf.PingPong(militaryTime - _lastUpdatedTime, 24f) >= MilitaryTimeDiffToUpdate)
		{
			_lastUpdatedTime = militaryTime;
			return true;
		}
		return false;
	}

	protected virtual void OnSunsetEvent()
	{
		SunsetEvent?.Invoke();

	private void Awake()
	{
		_sunOrbitRadius = _skyDome.transform.localScale.x * 0.99f;
		_inscatterEncoded = ConvertLookUpTable(_inscatter);
		_atmosphereLookupTexture = CreateBRDFTexture();
		_atmosphereLookupMaterial = CreateBRDFMaterial();
		SetStaticShaderUniforms();
		SetMaterialUniforms(_skyDome.sharedMaterial);
		RenderSettings.ambientMode = AmbientMode.Trilight;
		RenderSettings.ambientIntensity = 1f;
	}

	public RenderTexture CreateBRDFTexture()
	{
		RenderTexture renderTexture = (_atmosphereLookupTexture = new RenderTexture(512, 512, 0, RenderTextureFormat.ARGBHalf));
		renderTexture.filterMode = FilterMode.Bilinear;
		renderTexture.useMipMap = true;
		renderTexture.autoGenerateMips = true;
		return renderTexture;
	}

	public Material CreateBRDFMaterial()
	{
		Shader shader = Shader.Find("Beam Team/Atmosphere/Atmosphere BRDF");
		if (shader != null)
		{
			Material material = new Material(shader);
			SetMaterialUniforms(material);
			return material;
		}
		Debug.LogError("Atmosphere::Start: Atmosphere BRDF Shader could not be found.");
		return null;
	}

	private void Start()
	{
		Camera.onPreCull = (Camera.CameraCallback)Delegate.Combine(Camera.onPreCull, new Camera.CameraCallback(Camera_OnPreCull));
	}

	protected override void OnDestroy()
	{
		BeamTeam.Destroy(_atmosphereLookupMaterial);
		BeamTeam.Destroy(_atmosphereLookupTexture);
		BeamTeam.Destroy(_inscatterEncoded);
		Camera.onPreCull = (Camera.CameraCallback)Delegate.Remove(Camera.onPreCull, new Camera.CameraCallback(Camera_OnPreCull));
		base.OnDestroy();
	}

	private void Camera_OnPreCull(Camera camera)
	{
		if (!(camera == null) && (camera.CompareTag("MainCamera") || camera.CompareTag("PlayerCamera")))
		{
			ClampSkyPosition(camera.transform.position);
			CalculateIlluminaryColors(camera);
			CalculateAmbientColor(camera);
		}
	}

	public static Texture3D ConvertLookUpTable(Texture2D texture)
	{
		if ((bool)texture)
		{
			if (texture.height != Mathf.FloorToInt(Mathf.Sqrt(texture.width)))
			{
				Debug.LogWarning("Atmosphere::ConvertLookUpTable: The given 2D texture " + texture.name + " cannot be used as a 3D LUT.");
				return null;
			}
			int num = 16;
			int height = texture.height;
			int num2 = 16;
			Color[] pixels = texture.GetPixels();
			Color[] array = new Color[pixels.Length];
			for (int i = 0; i < num; i++)
			{
				for (int j = 0; j < height; j++)
				{
					for (int k = 0; k < num2; k++)
					{
						array[i + j * num + k * num * height] = pixels[i + j * num * num2 + k * num];
					}
				}
			}
			Texture3D texture3D = new Texture3D(num, height, num2, TextureFormat.ARGB32, mipChain: false);
			texture3D.wrapMode = TextureWrapMode.Clamp;
			texture3D.filterMode = FilterMode.Bilinear;
			texture3D.SetPixels(array);
			texture3D.Apply();
			return texture3D;
		}
		Debug.LogError("Atmosphere::ConvertLookUpTable(): Couldn't load 3D LUT texture. Texture3D will be disabled.");
		return null;
	}

	private void SetStaticShaderUniforms()
	{
		float value = (float)(new FastRandom(StrandedWorld.WORLD_SEED).NextDouble() * 1000.0);
		Shader.SetGlobalFloat("_RandomSeed", value);
		Shader.SetGlobalFloat("_SkyFade", 0f);
		Shader.SetGlobalFloat("_StormFade", 0f);
	}

	private void SetDynamicShaderUniforms()
	{
		Shader.SetGlobalColor("_SunColor", _sunIlluminary.Light.color.linear);
		Shader.SetGlobalFloat("_SunIntensity", _sunIlluminary.Light.intensity);
		Shader.SetGlobalVector("_SunPosition", _sunIlluminary.Transform.position);
		Shader.SetGlobalVector("_SunDirection", _sunIlluminary.Transform.forward * -1f);
		Shader.SetGlobalVector("_MoonDirection", _moonIlluminary.Transform.forward * -1f);
	}

	public void SetMaterialUniforms(Material material)
	{
		if (!(material == null))
		{
			material.SetTexture("_Transmittance", _transmittance);
			material.SetTexture("_Inscatter", _inscatterEncoded);
			material.SetTexture("_Irradiance", _irradiance);
			material.SetFloat("_MieG", _mieG);
			material.SetFloat("Rg", 6360000f);
			material.SetFloat("Rt", 6420000f);
			material.SetFloat("Rl", 6421000f);
			material.SetFloat("RES_R", 16f);
			material.SetFloat("RES_MU", 16f);
			material.SetFloat("RES_MU_S", 16f);
			material.SetFloat("RES_NU", 1f);
		}
	}

	private void ClampSkyPosition(Vector3 position)
	{
		Vector3 position2 = position;
		position2.y = 0f;
		transform.position = position2;
	}

	private void Update()
	{
		if (CheckUpdate())
		{
			CalculateIlluminaryPositions();
			SetDynamicShaderUniforms();
			SetMaterialUniforms(_skyDome.sharedMaterial);
			SetMaterialUniforms(_atmosphereLookupMaterial);
			ComputeAtmosphereLookupTexture(_atmosphereLookupMaterial, Singleton<GameTime>.Instance.NormalizedTime);
		}
	}

	public void ComputeAtmosphereLookupTexture(Material lookupMaterial, float normalizedTime)
	{
		lookupMaterial.SetColor("_PolarisingColor", _skyColorPolariser.Evaluate(normalizedTime));
		Graphics.Blit(null, _atmosphereLookupTexture, lookupMaterial);
		Shader.SetGlobalTexture("_SkyMap", _atmosphereLookupTexture);
	}

	private void CalculateIlluminaryPositions()
	{
		float f = MathF.PI / 180f * _latitude;
		float num = Mathf.Sin(f);
		float num2 = Mathf.Cos(f);
		float num3 = MathF.PI / 180f * _longitude;
		float f2 = 0.4093f * Mathf.Sin(MathF.PI / 184f * (_date - 81f));
		float num4 = Mathf.Sin(f2);
		float num5 = Mathf.Cos(f2);
		float num6 = (int)(_longitude / 15f);
		float num7 = MathF.PI / 12f * num6;
		float num8 = Singleton<GameTime>.Instance.MilitaryTime + 0.17f * Mathf.Sin(0.03369f * (_date - 80f)) - 0.129f * Mathf.Sin(MathF.PI * 2f / 355f * (_date - 8f)) + 12f / MathF.PI * (num7 - num3);
		float f3 = MathF.PI / 12f * num8;
		float num9 = Mathf.Sin(f3);
		float num10 = Mathf.Cos(f3);
		float num11 = Mathf.Asin(num * num4 - num2 * num5 * num10);
		float y = (0f - num5) * num9;
		float x = num2 * num4 - num * num5 * num10;
		float num12 = Mathf.Atan2(y, x);
		_theta = MathF.PI / 2f - num11;
		float phi = num12;
		_sunIlluminary.Transform.position = transform.position + transform.rotation * SphericalToCartesian(_sunOrbitRadius, _theta, phi);
		_sunIlluminary.Transform.LookAt(transform.position);
		_moonIlluminary.Transform.position = transform.position + transform.rotation * SphericalToCartesian(_sunOrbitRadius, _theta + MathF.PI, phi);
		_moonIlluminary.Transform.LookAt(transform.position);
		_altitude = MathF.PI / 2f - _theta;
		float num13 = Mathf.Abs(_altitude);
		float num14 = MathF.PI / 180f;
		float num15 = MathF.PI / 15f;
		if (_altitude > 0f)
		{
			_sunIlluminary.Enable();
			_moonIlluminary.Disable();
			float num16 = Mathf.Lerp(0f, _sunIlluminary.Intensity, num13 / num14);
			_sunIlluminary.CurrentIntensity = ((num13 < num14) ? num16 : _sunIlluminary.Intensity);
			_sunIlluminary.Light.intensity = _sunIlluminary.CurrentIntensity;
			if (!_dayTime)
			{
				_dayTime = true;
				OnSunsetEvent();
			}
		}
		else
		{
			_moonIlluminary.Enable();
			_sunIlluminary.Disable();
			float num17 = Mathf.Lerp(0f, _moonIlluminary.Intensity, num13 / num15);
			_moonIlluminary.CurrentIntensity = ((num13 < num15) ? num17 : _moonIlluminary.Intensity);
			_moonIlluminary.Light.intensity = _moonIlluminary.CurrentIntensity;
			if (_dayTime)
			{
				_dayTime = false;
				OnSunsetEvent();
			}
		}
	}

	private void CalculateIlluminaryColors(Camera camera)
	{
		bool flag = Cameras.IsUnderwater(camera.transform);
		float normalizedTime = Singleton<GameTime>.Instance.NormalizedTime;
		Color color = _sunIlluminary.Gradient.Evaluate(normalizedTime);
		Color color2 = _moonIlluminary.Gradient.Evaluate(normalizedTime);
		WeatherEvent weatherEvent = Singleton<AtmosphereStorm>.Instance?.CurrentWeatherEvent;
		if (weatherEvent != null && weatherEvent.HasStarted)
		{
			color = weatherEvent.GetSunColor(color);
			color2 = weatherEvent.GetMoonColor(color2);
		}
		_sunIlluminary.Light.color = color;
		_moonIlluminary.Light.color = color2;
		if (flag)
		{
			float num = Singleton<CetoSettings>.Instance.BelowVisibility * 1.5f;
			float num2 = Mathf.Clamp01((camera.transform.position.y + num) / num);
			CurrentIlluminary.Light.color = Color.Lerp(CurrentIlluminary.Light.color * new Color(0.5f, 0.9f, 1f), CurrentIlluminary.Light.color, num2);
			CurrentIlluminary.Light.intensity = CurrentIlluminary.CurrentIntensity * num2;
		}
	}

	private void CalculateAmbientColor(Camera camera)
	{
		bool flag = Cameras.IsUnderwater(camera.transform);
		float normalizedTime = Singleton<GameTime>.Instance.NormalizedTime;
		_ambientLightColor = _ambientSkyColorGradient.Evaluate(normalizedTime) * _skyColorPolariser.Evaluate(normalizedTime);
		_ambientGroundColor = (flag ? _ambientUnderwaterGroundColorGradient.Evaluate(normalizedTime) : _ambientGroundColorGradient.Evaluate(normalizedTime));
		WeatherEvent weatherEvent = Singleton<AtmosphereStorm>.Instance?.CurrentWeatherEvent;
		if (weatherEvent != null && weatherEvent.HasStarted)
		{
			_ambientLightColor = weatherEvent.GetAmbientSkyColor(_ambientLightColor);
			_ambientGroundColor = weatherEvent.GetAmbientGroundColor(_ambientGroundColor);
		}
		if (flag)
		{
			float num = Singleton<CetoSettings>.Instance.BelowVisibility * 1.5f;
			float num2 = Mathf.Clamp01((camera.transform.position.y + num) / num);
			_ambientLightColor *= num2;
			_ambientGroundColor *= num2;
		}
		RenderSettings.sun = CurrentIlluminary.Light;
		RenderSettings.ambientLight = _ambientLightColor;
		RenderSettings.ambientEquatorColor = Color.Lerp(_ambientGroundColor, _ambientLightColor, 0.6f);
		RenderSettings.ambientGroundColor = _ambientGroundColor;
	}

	private Vector3 SphericalToCartesian(float radius, float theta, float phi)
	{
		float num = Mathf.Sin(theta);
		float num2 = Mathf.Cos(theta);
		float num3 = Mathf.Sin(phi);
		float num4 = Mathf.Cos(phi);
		Vector3 zero = Vector3.zero;
		zero.x = radius * num * num4;
		zero.y = radius * num2;
		zero.z = radius * num * num3;
		return zero;
	}
}

using System;
using Beam;
using Beam.Events;
using Funlabs;
using UnityEngine;
using kode80.Clouds;

public class AtmosphereClouds : MonoBehaviour
{
	[SerializeField]
	private Shader _standard;

	[SerializeField]
	private Shader _volumetric;

	[SerializeField]
	private Renderer _cloudsRenderer;

	[SerializeField]
	private Gradient _cloudColor;

	[SerializeField]
	public Color _cloudBaseColorMultiplierBlendColor;

	[SerializeField]
	private float _cloudBaseColorMultiplier = 1f;

	public TimeOfDayKeyFrame[] keyFrames;

	[SerializeField]
	private Texture2D _overcastTexture;

	[SerializeField]
	private float _overcastTextureScale = 1f;

	[SerializeField]
	private float _overcastTextureSpeed = 0.2f;

	private Vector2 _overcastTextureDirection;

	private kode80Clouds _volumetricClouds;

	private bool _volumetricCloudsSupported;

	private void Start()
	{
		_volumetricClouds = UnityEngine.Object.FindObjectOfType<kode80Clouds>();
		_volumetricCloudsSupported = SystemInfo.graphicsDeviceVersion.StartsWith("Direct3D 11");
		if (_volumetricCloudsSupported)
		{
			Singleton<Atmosphere>.Instance.SunsetEvent += Atmosphere_SunsetEvent;
		}
		else
		{
			_volumetricClouds.gameObject.SetActive(value: false);
		}
		LoadOptions();
		EventManager.AddListener<OptionsAppliedEvent>(EventManager_OptionsApplied);
		SortKeyFrames();
	}

	private void OnDestroy()
	{
		EventManager.RemoveListener<OptionsAppliedEvent>(EventManager_OptionsApplied);
		if (_volumetricCloudsSupported && (bool)Singleton<Atmosphere>.Instance)
		{
			Singleton<Atmosphere>.Instance.SunsetEvent -= Atmosphere_SunsetEvent;
		}
	}

	private void EventManager_OptionsApplied(OptionsAppliedEvent e)
	{
		LoadOptions();
	}

	public void LoadOptions()
	{
		switch ((!Game.Mode.IsLocalCoop() && _volumetricCloudsSupported) ? Options.GraphicsSettings.Clouds : Quality_VolumetricClouds.OFF)
		{
		case Quality_VolumetricClouds.OFF:
			_volumetricClouds.enabled = false;
			_cloudsRenderer.sharedMaterial.shader = _standard;
			break;
		case Quality_VolumetricClouds.LOW:
			_volumetricClouds.enabled = true;
			_volumetricClouds.subPixelSize = kode80Clouds.SubPixelSize.Sub4x4;
			_volumetricClouds.maxIterations = 48;
			_cloudsRenderer.sharedMaterial.shader = _volumetric;
			break;
		case Quality_VolumetricClouds.MEDIUM:
			_volumetricClouds.enabled = true;
			_volumetricClouds.subPixelSize = kode80Clouds.SubPixelSize.Sub4x4;
			_volumetricClouds.maxIterations = 64;
			_cloudsRenderer.sharedMaterial.shader = _volumetric;
			break;
		case Quality_VolumetricClouds.HIGH:
			_volumetricClouds.enabled = true;
			_volumetricClouds.subPixelSize = kode80Clouds.SubPixelSize.Sub2x2;
			_volumetricClouds.maxIterations = 64;
			_cloudsRenderer.sharedMaterial.shader = _volumetric;
			break;
		}
	}

	private void Atmosphere_SunsetEvent()
	{
		_volumetricClouds.sunLight = (Singleton<Atmosphere>.Instance.DayTime ? Singleton<Atmosphere>.Instance.SunIlluminary.Light : Singleton<Atmosphere>.Instance.MoonIlluminary.Light);
	}

	private void Update()
	{
		Color value = _cloudColor.Evaluate(Singleton<GameTime>.Instance.NormalizedTime);
		_cloudsRenderer.sharedMaterial.SetColor("_CloudColor", value);
		UpdateVolumetricCloudsColors();
		UpdateVolumetricCloudsLighting();
		Vector2 vector = Wind.Direction;
		vector *= Time.timeSinceLevelLoad * _overcastTextureSpeed;
		Shader.SetGlobalVector("_OvercastTex_Coords", new Vector4(1f, 1f, vector.x, vector.y) * (1f / _overcastTextureScale));
		Shader.SetGlobalFloat("_Overcast", Mathf.Max(Mathf.Pow(Singleton<Atmosphere>.Instance.Altitude, 0.3f)));
		Shader.SetGlobalTexture("_OvercastTex", _overcastTexture ?? Texture2D.whiteTexture);
	}

	private void UpdateVolumetricCloudsColors()
	{
		if (_volumetricCloudsSupported && _volumetricClouds.enabled)
		{
			_volumetricClouds.cloudTopColor = _cloudColor.Evaluate(Singleton<GameTime>.Instance.NormalizedTime);
			_volumetricClouds.cloudBaseColor = UnityEngine.RenderSettings.ambientSkyColor * (_cloudBaseColorMultiplierBlendColor * _cloudBaseColorMultiplier);
		}
	}

	private void UpdateVolumetricCloudsLighting()
	{
		if (_volumetricCloudsSupported && _volumetricClouds.enabled && keyFrames.Length >= 1)
		{
			float x = Singleton<Atmosphere>.Instance.SunIlluminary.Transform.localEulerAngles.x;
			TimeOfDayKeyFrame timeOfDayKeyFrame = KeyFrameBelowAngle(x);
			TimeOfDayKeyFrame timeOfDayKeyFrame2 = KeyFrameAboveAngle(x);
			float num = timeOfDayKeyFrame2.angle - timeOfDayKeyFrame.angle;
			float t = ((num == 0f) ? 0.5f : ((x - timeOfDayKeyFrame.angle) / num));
			_volumetricClouds.sunScalar = Mathf.Lerp(timeOfDayKeyFrame.SunScalar, timeOfDayKeyFrame2.SunScalar, t);
			_volumetricClouds.ambientScalar = Mathf.Lerp(timeOfDayKeyFrame.AmbientScalar, timeOfDayKeyFrame2.AmbientScalar, t);
		}
	}

	public void SortKeyFrames()
	{
		Array.Sort(keyFrames, (TimeOfDayKeyFrame a, TimeOfDayKeyFrame b) => a.angle.CompareTo(b.angle));
	}

	private TimeOfDayKeyFrame KeyFrameBelowAngle(float angle)
	{
		return KeyFrameForAngle(angle, findGreaterThan: false);
	}

	private TimeOfDayKeyFrame KeyFrameAboveAngle(float angle)
	{
		return KeyFrameForAngle(angle, findGreaterThan: true);
	}

	private TimeOfDayKeyFrame KeyFrameForAngle(float angle, bool findGreaterThan)
	{
		if (keyFrames.Length != 0)
		{
			TimeOfDayKeyFrame timeOfDayKeyFrame = keyFrames[0];
			TimeOfDayKeyFrame timeOfDayKeyFrame2 = keyFrames[keyFrames.Length - 1];
			if (angle <= timeOfDayKeyFrame.angle)
			{
				return timeOfDayKeyFrame;
			}
			if (angle >= timeOfDayKeyFrame2.angle)
			{
				return timeOfDayKeyFrame2;
			}
			for (int i = 0; i < keyFrames.Length; i++)
			{
				timeOfDayKeyFrame = keyFrames[i];
				timeOfDayKeyFrame2 = ((i < keyFrames.Length - 1) ? keyFrames[i + 1] : timeOfDayKeyFrame);
				if (angle >= timeOfDayKeyFrame.angle && angle <= timeOfDayKeyFrame2.angle)
				{
					if (!findGreaterThan)
					{
						return timeOfDayKeyFrame;
					}
					return timeOfDayKeyFrame2;
				}
			}
		}
		return null;
	}
}

	[Range(0f, 100f)]
	[SerializeField]

	public event WeatherEventHandler WeatherStarted;

	public event EmptyEventHandler WeatherStopped;

	protected virtual void OnWeatherStarted(WeatherEvent weather)

		WeatherStarted?.Invoke(weather);
	}

	protected virtual void OnWeatherStopped()

	private void Start()
	{
		Shader.SetGlobalFloat("_SheetLightning", 0f);
		Singleton<GameTime>.Instance.NewDayTick += GameTime_NewDayTick;
		Singleton<GameTime>.Instance.HourChanged += GameTime_HourChanged;
		for (int i = 0; i < _weatherEvents.Count; i++)
		{
			_weatherEvents[i].Started += WeatherEvent_Started;
			_weatherEvents[i].Stopped += WeatherEvent_Stopped;
		}
		EventManager.AddListener<EndGameSequenceEvent>(EventManager_EndGameSequenceEvent);
		if (Game.Mode.IsMaster())
		{
			SetState(StrandedWorld.WORLD_SEED, 0);
		}
	}

	protected override void OnDestroy()
	{
		if ((bool)Singleton<GameTime>.Instance)
		{
			Singleton<GameTime>.Instance.NewDayTick -= GameTime_NewDayTick;
			Singleton<GameTime>.Instance.HourChanged -= GameTime_HourChanged;
		}
		for (int i = 0; i < _weatherEvents.Count; i++)
		{
			_weatherEvents[i].Started -= WeatherEvent_Started;
			_weatherEvents[i].Stopped -= WeatherEvent_Stopped;
		}
		EventManager.RemoveListener<EndGameSequenceEvent>(EventManager_EndGameSequenceEvent);
		WeatherStarted = null;
		WeatherStopped = null;
		base.OnDestroy();
	}

	private void SetState(int seed, int iter)
	{
		_random.RestoreState(seed, iter);
	}

	private void EventManager_EndGameSequenceEvent(EndGameSequenceEvent ev)
	{
		ClearWeatherEvent();
		_canDoWeather = false;
	}

	private void GameTime_NewDayTick()
	{
		if (!Game.Mode.IsClient())
		{
			ReplicatedCheckWeather();
		}
	}

	private void GameTime_HourChanged(float hours)
	{
		if (!(_currentWeatherEvent != null))
		{
			return;
		}
		if (_currentWeatherEvent.HasStarted)
		{
			if (GameTime.GetMinutesSince(_currentWeatherEvent.StartedTime) > _currentWeatherEvent.TotalEventMinutes)
			{
				_currentWeatherEvent.ResetEvent();
			}
		}
		else if (Singleton<GameTime>.Instance.MilitaryTime > _currentWeatherEvent.StartTime + _currentWeatherEvent.TotalEventHours)
		{
			ClearWeatherEvent();
		}
		else if (Singleton<GameTime>.Instance.MilitaryTime > _currentWeatherEvent.StartTime)
		{
			float hours2 = Singleton<GameTime>.Instance.MilitaryTime - _currentWeatherEvent.StartTime;
			_currentWeatherEvent.StartedTime = GameTime.Now.RemoveHours(hours2);
			_currentWeatherEvent.ResumeEvent();
			CancelInvoke("PollTimeForWeatherEvent");
		}
	}

	private void CheckWeather()
	{
		if (_lastDayChecked == Singleton<GameCalendar>.Instance.Day)
		{
			return;
		}
		if (_lastRainyDay != Singleton<GameCalendar>.Instance.Day)
		{
			if (_currentWeatherEvent != null)
			{
				_lastRainyDay = Singleton<GameCalendar>.Instance.Day;
			}
			else if ((float)_random.NextDouble() * 100f <= (float)_percentage)
			{
				_lastRainyDay = Singleton<GameCalendar>.Instance.Day;
				CreateWeatherEvent();
			}
		}
		_lastDayChecked = Singleton<GameCalendar>.Instance.Day;
	}

	private void CreateWeatherEvent()
	{
		List<WeatherEvent> list = new List<WeatherEvent>();
		for (int i = 0; i < _weatherEvents.Count; i++)
		{
			WeatherEvent weatherEvent = _weatherEvents[i];
			for (int j = 0; j < weatherEvent.Percentage / 10; j++)
			{
				list.Add(weatherEvent);
			}
		}
		int index = _random.Next(0, list.Count);
		WeatherEvent weatherEvent2 = list[index];
		float startTime = 10f + 6f * (float)_random.NextDouble();
		weatherEvent2.StartTime = startTime;
		NewWeatherEvent(weatherEvent2);
	}

	private void PollTimeForWeatherEvent()
	{
		if (_canDoWeather && Singleton<GameTime>.Instance.MilitaryTime >= _currentWeatherEvent.StartTime)
		{
			CancelInvoke("PollTimeForWeatherEvent");
			StartWeatherEvent();
		}
	}

	public void StartWeatherEvent()
	{
		if (_currentWeatherEvent != null)
		{
			_currentWeatherEvent.StartEvent();
			if (Mathf.Approximately(_currentWeatherEvent.Humidity, 100f))
			{
				_rain++;
			}
		}
	}

	private void WeatherEvent_Started(WeatherEvent weather)
	{
		OnWeatherStarted(weather);
	}

	private void WeatherEvent_Stopped()
	{
		OnWeatherStopped();
	}

	public void NewWeatherEvent(WeatherEvent weather)
	{
		SetWeatherEvent(weather);
		InvokeRepeating("PollTimeForWeatherEvent", 0f, 2f);
	}

	public void SetWeatherEvent(WeatherEvent weather)
	{
		_currentWeatherEvent = weather;
	}

	public void ClearWeatherEvent()
	{
		_currentWeatherEvent = null;
		CancelInvoke("PollTimeForWeatherEvent");
	}

	private void ReplicatedCheckWeather()
	{
		if (Game.Mode.IsMultiplayer())
		{
			ReplicateCheckWeatherMsg replicateCheckWeatherMsg = new ReplicateCheckWeatherMsg();
			replicateCheckWeatherMsg.Seed = _random.GetSeed();
			replicateCheckWeatherMsg.Iter = _random.GetIter();
			replicateCheckWeatherMsg.Post();
		}
		CheckWeather();
	}

	public JObject Save()
	{
		JObject jObject = new JObject();
		jObject.AddField("Rain", _rain);
		JObject data = SavingUtilities.Save(_random);
		jObject.AddField("State", data);
		if (CurrentWeatherEvent != null)
		{
			JObject data2 = CurrentWeatherEvent.Save();
			jObject.AddField(CurrentWeatherEvent.gameObject.name, data2);
		}
		return jObject;
	}

	public void Load(JObject data)
	{
		_rain = data.GetField("Rain").GetValue<int>();
		JObject field = data.GetField("State");
		if ((bool)field)
		{
			SavingUtilities.Load(_random, field);
		}
		for (int i = 0; i < _weatherEvents.Count; i++)
		{
			WeatherEvent weatherEvent = _weatherEvents[i];
			JObject field2 = data.GetField(weatherEvent.name);
			if (field2 != null)
			{
				weatherEvent.Load(field2);
				break;
			}
		}
	}
}
